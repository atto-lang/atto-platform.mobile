cmake_minimum_required(VERSION 3.20)
project(ores_phone_policy VERSION 0.1.0 LANGUAGES C)
include(CTest)

add_library(ores_phone_policy STATIC src/ores_phone_policy.c)
target_include_directories(ores_phone_policy PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_features(ores_phone_policy PUBLIC c_std_11)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(ores_phone_policy PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
if(BUILD_TESTING AND NOT CMAKE_CROSSCOMPILING)
  add_executable(test_phone_policy tests/test_phone_policy.c)
  target_link_libraries(test_phone_policy PRIVATE ores_phone_policy)
  if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(test_phone_policy PRIVATE -Wall -Wextra -Wpedantic -Werror -UNDEBUG)
  endif()
  add_test(NAME phone_policy_contract COMMAND test_phone_policy)
endif()
